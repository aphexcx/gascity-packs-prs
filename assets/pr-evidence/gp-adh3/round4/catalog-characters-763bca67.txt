# check_chars.py
"""Each catalog row's emoji, against Slack's own table at the commit the adapter's name list pins."""
import json, sys
table = json.load(open(sys.argv[1], encoding="utf-8"))
by = {n: e for e in table for n in e.get("short_names", [])}
rows = json.load(open(sys.argv[2], encoding="utf-8"))["reactions"]
bad = 0
for r in rows:
    e = by[r["slack_name"]]
    ch = "".join(chr(int(h, 16)) for h in e["unified"].split("-"))
    ok = ch == r["emoji"]
    bad += not ok
    cps = " ".join(f"U+{ord(c):04X}" for c in r["emoji"])
    print(f'{"ok  " if ok else "BAD "} {r["slack_name"]:17} {r["emoji"]}  {cps:15} Slack table: {e["unified"]:10} {e["name"]}')
print(f"{len(rows)} rows, {bad} differ")
sys.exit(1 if bad else 0)
