async function owners(...ids: string[]) {
  const { results } = await env.DB.prepare(
    `SELECT conversation_id, wa_id, user_id, city_id FROM phones
     WHERE conversation_id IN (SELECT value FROM json_each(?1)) OR wa_id IN (SELECT value FROM json_each(?1))
        OR user_id IN (SELECT value FROM json_each(?1))
     ORDER BY bound_at, conversation_id`,
  )
    .bind(JSON.stringify(ids))
    .all();
  return results;
}
type T = ReturnType<typeof setup>;
const wamidOf = (p: any): string => p.entry[0].changes[0].value.messages[0].id;
const ids = async (t: T, city: City) => (await t.inbox(city)).body.deliveries.map((d: any) => d.message?.id);
const imageAt = (t: T, from: string, ts: number) => {
  const mediaId = `7${ts}${Math.floor(Math.random() * 1_000_000)}`;
  t.graph.media.set(mediaId, { mime: "image/jpeg", bytes: new Uint8Array([255, 216, 255]) });
  return { payload: mediaPayload("image", from, mediaId, { wamid: `wamid.${uniq()}`, ts }), mediaId };
};
const mediaStatus = async (t: T, city: City, mediaId: string) => (await t.call("GET", `/v1/media/${mediaId}`, { token: city.token })).status;
const stat = async (name: string) =>
  (await env.DB.prepare("SELECT value FROM stats WHERE name = ?").bind(name).first<{ value: number }>())?.value ?? 0;
const unbind = (conversationId: string) => env.DB.prepare("DELETE FROM phones WHERE conversation_id = ?1").bind(conversationId).run();
const spans = async (id: string) => {
  try {
    return (await env.DB.prepare("SELECT conversation_id, since, changed_at, routes_from FROM phone_aliases WHERE id = ? ORDER BY since").bind(id).all()).results;
  } catch {
    return (await env.DB.prepare("SELECT conversation_id, since, changed_at FROM phone_aliases WHERE id = ? ORDER BY since").bind(id).all()).results;
  }
};
const notice = (from: string | null, previousUser: string, newUser: string, newWa: string | null, ts: number) =>
  changeAs(newWa ? "user_changed_number" : "user_changed_user_id", { from, previousUser, newUser, newWa }, { wamid: `wamid.${uniq()}`, ts });

/** Where one image went: per city, the delivery and the media grant; the counts; Graph sends. */
async function landed(t: T, cities: Record<string, City>, image: { payload: any; mediaId: string }) {
  const before = { unrouted: await stat("unrouted_messages"), conflicts: await stat("unrouted_conflicts"), sends: t.graph.sends().length };
  const status = (await t.webhook(image.payload)).status;
  const out: Record<string, unknown> = { status };
  for (const [name, city] of Object.entries(cities)) {
    out[name] = { delivered: (await ids(t, city)).includes(wamidOf(image.payload)), media: await mediaStatus(t, city, image.mediaId) };
  }
  out.unrouted = (await stat("unrouted_messages")) - before.unrouted;
  out.conflicts = (await stat("unrouted_conflicts")) - before.conflicts;
  out.sends = t.graph.sends().length - before.sends;
  return out;
}

    const t = setup({ db: flaky.db });
    const x = await t.enrollCity();
    const w = await t.enrollCity();
    const p = phone();
    await t.join(w, p); // C
    const a = bsuid();
    const k = await t.invite(x);
    const text = textAs({ wa: p, user: a }, k, { wamid: `wamid.${uniq()}`, ts: T1 });
    t.clock.now = T1;
    flaky.arm(0);
    expect((await t.webhook(text)).status).toBe(500);
    t.clock.now = T1 + HOUR / 4;
    const k2 = await t.invite(x);
    t.clock.now = T2;
    await t.sayAs({ wa: p, user: a }, k2); // binds a to X (P contested)
    t.clock.now = T2 + HOUR / 4;
    expect((await t.webhook(text)).status).toBe(200); // Meta's retry of the K text
    const out = { inW: (await ids(t, w)).includes(wamidOf(text)), inX: (await ids(t, x)).includes(wamidOf(text)), owners: await owners(p, a) };
    console.log("R8", JSON.stringify(out));
    expect(out.inW).toBe(false);
  });

  /*
   * R9 (no 500 needed): a, beside phone-only C's P, first mistypes X's code (a six-digit text that
   * is no live code: rule 3, C adopts a), then sends X's live code. The live code is now a known
   * sender's (C's) text: it reaches W, stays unspent, and a can never bind to X.
   */
  it("R9: a typo'd code before the live one: X's live code text must not reach C's city, and C's row must not adopt a", async () => {
    const t = setup();
    const x = await t.enrollCity();
    const w = await t.enrollCity();
    const p = phone();
    await t.join(w, p); // C
    const c = (await owners(p))[0] as { conversation_id: string };
    const a = bsuid();
    const k = await t.invite(x);
    const typo = k === "000000" ? "000001" : "000000";
    t.clock.now = T1;
    const first = await t.sayAs({ wa: p, user: a }, typo);
    t.clock.now = T1 + 60;
    const live = await t.sayAs({ wa: p, user: a }, k);
    const used = (await env.DB.prepare("SELECT used_at FROM invite_codes WHERE code = ?").bind(k).first<{ used_at: number | null }>())?.used_at;
    const out = {
      typoInW: (await ids(t, w)).includes(first.wamid),
      liveInW: (await ids(t, w)).includes(live.wamid),
      liveInX: (await ids(t, x)).includes(live.wamid),
      codeUsedAt: used,
      owners: await owners(p, a),
    };
    console.log("R9", JSON.stringify(out), "C", c.conversation_id);
    expect(out.liveInW).toBe(false);
    expect(out.owners).not.toContainEqual(expect.objectContaining({ conversation_id: c.conversation_id, user_id: a }));
  });

  /*
   * R6 (the fill is late evidence that routes; pre-existing): A by user_id a alone; A's (P, a)
   * stamped T1 fails; B pairs P at T2 (B's row takes P) and leaves it at T3 (B's notice); A's T1
   * text lands after: no row holds P, so A's row takes P with wa_since = T1. An image from P stamped
   * after B left P (not A's: P was B's from T2) routes to A's city.
   */
  it("R6: A's late (P, a) text fills A's row with P from T1 after B held P [T2, T3]; an image from P stamped after T3 must not reach A's city", async () => {
    const flaky = flakyDb();
    const t = setup({ db: flaky.db });
    const x = await t.enrollCity();
    const y = await t.enrollCity();
    const p = phone();
    const a = bsuid();
    await t.joinAs(x, { user: a });
    const b = bsuid();
    await t.joinAs(y, { user: b });
    const aLate = textAs({ wa: p, user: a }, "A at T1", { wamid: `wamid.${uniq()}`, ts: T1 });
    t.clock.now = T1;
    flaky.arm(0);
    expect((await t.webhook(aLate)).status).toBe(500);
    t.clock.now = T2;
    await t.sayAs({ wa: p, user: b }, "B at T2"); // B's row takes P
    t.clock.now = T3;
    expect((await t.webhook(notice(p, b, bsuid(), phone(), T3))).status).toBe(200); // B leaves P
    t.clock.now = T3 + HOUR / 4;
    expect((await t.webhook(aLate)).status).toBe(200);
    console.log("R6 A's row", JSON.stringify(await env.DB.prepare("SELECT wa_id, wa_since FROM phones WHERE user_id = ?").bind(a).first()));
    const image = imageAt(t, p, T4);
    t.clock.now = T4 + HOUR / 4;
    const out = await landed(t, { x, y }, image);
    console.log("R6 landed", JSON.stringify(out));
    expect(out).toMatchObject({ status: 200, x: { delivered: false, media: 404 } });
  });
